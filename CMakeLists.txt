cmake_minimum_required(VERSION 3.24)
project(CppLessons LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_SANITIZERS "Enable ASan and UBSan for tests" OFF)

enable_testing()

file(GLOB_RECURSE TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*_test.cpp")
foreach(test_src ${TEST_SOURCES})
  get_filename_component(test_name ${test_src} NAME_WE)
  add_executable(${test_name} ${test_src})
  target_include_directories(${test_name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/lessons")
  if(ENABLE_SANITIZERS)
    target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wpedantic -Werror -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${test_name} PRIVATE -fsanitize=address,undefined)
  else()
    target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wpedantic -Werror -O2)
  endif()
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

function(add_lesson_benchmark target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/lessons")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -O2)
endfunction()

file(GLOB BENCH_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lessons/*/benchmark.cpp")
foreach(bench_src ${BENCH_SOURCES})
  get_filename_component(bench_dir ${bench_src} DIRECTORY)
  get_filename_component(lesson_name ${bench_dir} NAME)
  add_lesson_benchmark(${lesson_name}_benchmark ${bench_src})
endforeach()
